mostrar("1. Ventas entregadas por método de pago", db.pedidos.aggregate([
  { $match: { estado: "entregado" } },
  { $group: { _id: "$metodo_pago", ventas: { $sum: "$total" }, pedidos: { $sum: 1 } } },
  { $sort: { ventas: -1 } }
]).toArray());


// 2. Cuántos pedidos se cancelaron
mostrar("2. Pedidos cancelados", db.pedidos.countDocuments({ estado: "cancelado" }));


// 3. Menú disponible de cada restaurante: lista de productos, precio máximo,

mostrar("3. Menú disponible por restaurante (primeros 3 por id)", db.productos.aggregate([
  { $match: { disponible: true } },
  { $sort: { precio: -1 } },                       // la lista queda del más caro al más barato
  { $group: {
      _id: "$restaurante_id",
      menu: { $push: { nombre: "$nombre", precio: "$precio" } },
      precio_max: { $max: "$precio" }
  } },
  { $sort: { _id: 1 } },
  { $limit: 3 }
]).toArray());


// 3b. Variante: TODOS los restaurantes, pero solo sus 3 productos más caros
mostrar("3b. Top 3 productos más caros por restaurante", db.productos.aggregate([
  { $match: { disponible: true } },
  { $sort: { precio: -1 } },
  { $group: {
      _id: "$restaurante_id",
      menu: { $push: { nombre: "$nombre", precio: "$precio" } },
      precio_max: { $max: "$precio" }
  } },
  { $project: { precio_max: 1, menu: { $slice: ["$menu", 3] } } },
  { $sort: { _id: 1 } }
]).toArray());

// 4. Productos por tipo con precio promedio
mostrar("4. Productos por tipo", db.productos.aggregate([
  { $group: { _id: "$tipo", productos: { $sum: 1 }, precio_promedio: { $avg: "$precio" } } },
  { $project: { productos: 1, precio_promedio: { $round: ["$precio_promedio", 2] } } },
  { $sort: { precio_promedio: -1 } }
]).toArray());


// 5. Los 3 clientes con mayor gasto en pedidos entregados
mostrar("5. Top 3 clientes por gasto", db.pedidos.aggregate([
  { $match: { estado: "entregado" } },
  { $group: { _id: "$cliente.id", nombre: { $first: "$cliente.nombre" }, gasto_total: { $sum: "$total" }, pedidos: { $sum: 1 } } },
  { $sort: { gasto_total: -1 } },
  { $limit: 3 }
]).toArray());


// 6. Restaurantes con ventas entregadas superiores a 70.000
mostrar("6. Restaurantes con ventas > 70000", db.pedidos.aggregate([
  { $match: { estado: "entregado" } },
  { $group: { _id: "$restaurante.id", nombre: { $first: "$restaurante.nombre" }, ventas: { $sum: "$total" } } },
  { $match: { ventas: { $gt: 70000 } } },          // este $match va DESPUÉS del $group: filtra sobre la suma
  { $sort: { ventas: -1 } }
]).toArray());


// 7. Pedidos pendientes o en camino con el email del cliente para notificarlo
mostrar("7. Pedidos por notificar", db.pedidos.aggregate([
  { $match: { estado: { $in: ["pendiente", "en_camino"] } } },
  { $lookup: { from: "clientes", localField: "cliente.id", foreignField: "_id", as: "datos_cliente" } },
  { $unwind: "$datos_cliente" },
  { $project: {
      estado: 1,
      total: 1,
      cliente: "$cliente.nombre",
      email: "$datos_cliente.email"
  } },
  { $sort: { _id: 1 } }
]).toArray());

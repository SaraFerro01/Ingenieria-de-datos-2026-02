db.estudiante.aggregate([
  { $match: { activo: true } },
  {
    $lookup: {
      from: "curso",
      localField: "idcurso",
      foreignField: "_id",
      as: "curso_info"
    }
  },
  { $unwind: "$curso_info" },
  {
    $lookup: {
      from: "nivel",
      localField: "curso_info.idnivel",
      foreignField: "_id",
      as: "nivel_info"
    }
  },
  { $unwind: "$nivel_info" },
  {
    $lookup: {
      from: "jornada",
      localField: "curso_info.idjornada",
      foreignField: "_id",
      as: "jornada_info"
    }
  },
  { $unwind: "$jornada_info" },
  {
    $project: {
      tipodocumentoEst: 1,
      numerodocumentoEst: 1,
      nombres: 1,
      apellidos: 1,
      curso: "$curso_info.nombre",
      grado: "$curso_info.grado",
      nivel: "$nivel_info.nombre",
      jornada: "$jornada_info.nombre"
    }
  }
]);

db.asistencia.aggregate([
  { 
    $match: { 
      categoria: "FALTO",
      fecha: { $gte: ISODate("2026-08-01T00:00:00Z"), $lte: ISODate("2026-08-31T23:59:59Z") }
    } 
  },
  {
    $lookup: {
      from: "curso",
      localField: "idcurso",
      foreignField: "_id",
      as: "curso_info"
    }
  },
  { $unwind: "$curso_info" },
  {
    $group: {
      _id: "$idcurso",
      curso: { $first: "$curso_info.nombre" },
      inasistencias: { $sum: 1 }
    }
  },
  { $sort: { inasistencias: -1 } }
]);

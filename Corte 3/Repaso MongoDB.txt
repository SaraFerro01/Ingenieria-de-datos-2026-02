use saborya

const esquemaRestaurantes = {
  bsonType: "object",
  required: ["nombre", "categoria", "ciudad"],
  properties: {
    nombre:       { bsonType: "string", minLength: 2 },
    categoria:    { bsonType: "string" },
    ciudad:       { bsonType: "string" },
    calificacion: { bsonType: ["double", "int"], minimum: 0, maximum: 5, description: "número entre 0 y 5" },
    domicilio:    { bsonType: "bool" },
    etiquetas:    { bsonType: "array", items: { bsonType: "string" } },
    ubicacion: { bsonType: "object", required: ["type", "coordinates"],
      properties: { type: { enum: ["Point"] },
                    coordinates: { bsonType: "array", minItems: 2, maxItems: 2, items: { bsonType: ["double", "int"] } } } }
  }
}
db.restaurantes.find({ $nor: [ { $jsonSchema: esquemaRestaurantes } ] }, { nombre: 1, calificacion: 1 })

db.runCommand({ collMod: "restaurantes", validator: { $jsonSchema: esquemaRestaurantes },
                validationLevel: "strict", validationAction: "error" })

db.restaurantes.updateOne({ nombre: "Pizza Rápida" }, { $set: { calificacion: "cinco" } })   // rechazado
db.restaurantes.updateOne({ nombre: "Pizza Rápida" },
  { $set: { categoria: "Italiana", calificacion: 4, domicilio: true } })                      // aceptado
db.restaurantes.find({ $nor: [ { $jsonSchema: esquemaRestaurantes } ] })                       // sin resultados

db.runCommand({ collMod: "clientes", validator: { $jsonSchema: {
  bsonType: "object", required: ["nombre", "email"],
  properties: {
    nombre: { bsonType: "string" },
    email:  { bsonType: "string", pattern: "^[^@\\s]+@[^@\\s]+\\.[^@\\s]+$" },
    puntos: { bsonType: ["int", "long"], minimum: 0 },
    direcciones: { bsonType: "array", items: { bsonType: "object", required: ["direccion"] } }
  } } } })

db.createCollection("productos", { validator: { $jsonSchema: {
  bsonType: "object", required: ["restaurante_id", "nombre", "precio", "stock"],
  properties: {
    restaurante_id: { bsonType: ["objectId", "string"] },
    nombre:     { bsonType: "string" },
    precio:     { bsonType: ["int", "long", "decimal"], minimum: 0, description: "pesos como número entero" },
    stock:      { bsonType: ["int", "long"], minimum: 0 },
    disponible: { bsonType: "bool" }
  } } } })

db.createCollection("pedidos", { validator: { $jsonSchema: {
  bsonType: "object", required: ["cliente", "restaurante", "fecha", "estado", "items", "total"],
  properties: {
    cliente:     { bsonType: "object", required: ["id", "nombre"] },
    restaurante: { bsonType: "object", required: ["id", "nombre", "ciudad"] },
    fecha:       { bsonType: "date" },
    estado:      { enum: ["pendiente", "en_camino", "entregado", "cancelado"] },
    metodo_pago: { enum: ["tarjeta", "efectivo", "nequi"] },
    items: { bsonType: "array", minItems: 1, items: { bsonType: "object",
      required: ["producto_id", "nombre", "precio", "cantidad"],
      properties: { precio: { bsonType: ["int", "long", "decimal"], minimum: 0 },
                    cantidad: { bsonType: ["int", "long"], minimum: 1 } } } },
    total: { bsonType: ["int", "long", "decimal"], minimum: 0 }
  } } } })
